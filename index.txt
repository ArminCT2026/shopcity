<script>
const ADMIN_PASSWORD = "1234";

const starterProducts = [
 {id:1,name:"هدفون بی‌سیم",price:1250000,category:"دیجیتال",emoji:"🎧"},
 {id:2,name:"ساعت هوشمند",price:2100000,category:"دیجیتال",emoji:"⌚"},
 {id:3,name:"کیبورد گیمینگ",price:1750000,category:"گیمینگ",emoji:"⌨️"},
 {id:4,name:"کفش اسپرت",price:980000,category:"پوشاک",emoji:"👟"},
 {id:5,name:"چراغ مطالعه",price:450000,category:"خانه",emoji:"💡"},
 {id:6,name:"دسته بازی",price:1450000,category:"گیمینگ",emoji:"🎮"}
];

function loadData(key,fallback){
 try{
  const value=localStorage.getItem(key);
  return value===null?fallback:JSON.parse(value);
 }catch(e){return fallback}
}

let products=loadData("sc_products",starterProducts);
let cart=loadData("sc_cart",[]);
let activeCategory="همه";

function save(){
 localStorage.setItem("sc_products",JSON.stringify(products));
 localStorage.setItem("sc_cart",JSON.stringify(cart));
}

function escapeHTML(value){
 return String(value).replace(/[&<>"']/g,c=>({
  "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
 })[c]);
}

function money(value){
 return Number(value).toLocaleString("fa-IR")+" تومان";
}

function renderCategories(){
 const categories=["همه",...new Set(products.map(p=>p.category))];
 document.getElementById("categories").innerHTML=categories.map(c=>
  `<button class="category ${activeCategory===c?'active':''}"
   onclick="selectCategory(${JSON.stringify(c)})">${escapeHTML(c)}</button>`
 ).join("");
}

function selectCategory(category){
 activeCategory=category;
 renderCategories();
 renderProducts();
}

function renderProducts(){
 const search=document.getElementById("searchInput").value.trim();
 const filtered=products.filter(p=>
  (activeCategory==="همه"||p.category===activeCategory)&&
  p.name.includes(search)
 );

 document.getElementById("productCount").textContent=filtered.length+" محصول";

 document.getElementById("products").innerHTML=filtered.map(p=>`
  <article class="product">
   <div class="product-image">${escapeHTML(p.emoji)}</div>
   <small>${escapeHTML(p.category)}</small>
   <h3>${escapeHTML(p.name)}</h3>
   <div class="price">${money(p.price)}</div>
   <button class="btn add" onclick="addToCart(${p.id})">
    افزودن به سبد خرید
   </button>
  </article>
 `).join("")||"<p>محصولی پیدا نشد.</p>";
}

function addProduct(){
 const name=document.getElementById("newName").value.trim();
 const price=Number(document.getElementById("newPrice").value);
 const category=document.getElementById("newCategory").value;
 const emoji=document.getElementById("newEmoji").value.trim()||"📦";

 if(!name||!Number.isFinite(price)||price<=0){
  alert("نام و قیمت معتبر وارد کن.");
  return;
 }

 products.push({id:Date.now(),name,price,category,emoji});
 save();

 document.getElementById("newName").value="";
 document.getElementById("newPrice").value="";
 document.getElementById("newEmoji").value="";

 renderAll();
}

function deleteProduct(id){
 products=products.filter(p=>p.id!==id);
 cart=cart.filter(item=>item.id!==id);
 save();
 renderAll();
}

function renderAdmin(){
 document.getElementById("adminProducts").innerHTML=products.map(p=>`
  <div class="admin-item">
   <span>${escapeHTML(p.emoji)} ${escapeHTML(p.name)} — ${money(p.price)}</span>
   <button class="delete" onclick="deleteProduct(${p.id})">حذف</button>
  </div>
 `).join("")||"<p>محصولی ثبت نشده.</p>";
}

function addToCart(id){
 const item=cart.find(p=>p.id===id);
 if(item)item.quantity++;
 else cart.push({id,quantity:1});
 save();
 renderCart();
}

function renderCart(){
 const count=cart.reduce((sum,item)=>sum+item.quantity,0);
 document.getElementById("cartCount").textContent=count;
 let total=0;

 document.getElementById("cartItems").innerHTML=cart.map(item=>{
  const p=products.find(x=>x.id===item.id);
  if(!p)return "";
  total+=p.price*item.quantity;
  return `<div class="cart-row">
   <span>${escapeHTML(p.name)} × ${item.quantity}</span>
   <button class="delete" onclick="removeCart(${p.id})">حذف</button>
  </div>`;
 }).join("")||"<p>سبد خرید خالی است.</p>";

 document.getElementById("totalPrice").textContent="مجموع: "+money(total);
}

function removeCart(id){
 cart=cart.filter(item=>item.id!==id);
 save();
 renderCart();
}

function toggleCart(){
 document.getElementById("cartBox").classList.toggle("show");
}

function toggleAdmin(){
 const panel=document.getElementById("adminPanel");
 panel.classList.toggle("show");

 if(panel.classList.contains("show")){
  if(sessionStorage.getItem("sc_admin_logged_in")==="true"){
   showAdminContent();
  }else{
   document.getElementById("adminLogin").style.display="block";
   document.getElementById("adminContent").style.display="none";
  }
 }
}

function loginAdmin(){
 const password=document.getElementById("adminPassword").value;

 if(password===ADMIN_PASSWORD){
  sessionStorage.setItem("sc_admin_logged_in","true");
  showAdminContent();
  document.getElementById("adminPassword").value="";
  document.getElementById("loginError").textContent="";
 }else{
  document.getElementById("loginError").textContent="رمز عبور اشتباه است!";
 }
}

function showAdminContent(){
 document.getElementById("adminLogin").style.display="none";
 document.getElementById("adminContent").style.display="block";
 renderAdmin();
}

function logoutAdmin(){
 sessionStorage.removeItem("sc_admin_logged_in");
 document.getElementById("adminContent").style.display="none";
 document.getElementById("adminLogin").style.display="block";
 document.getElementById("adminPassword").value="";
}

function toggleTheme(){
 document.body.classList.toggle("dark");
 localStorage.setItem("sc_dark",document.body.classList.contains("dark"));
}

function changeColor(color){
 document.documentElement.style.setProperty("--main",color);
 localStorage.setItem("sc_color",color);
}

function resetProducts(){
 if(!confirm("محصولات به حالت اولیه برگردند؟"))return;
 products=starterProducts.map(p=>({...p}));
 save();
 renderAll();
}

function renderAll(){
 renderCategories();
 renderProducts();
 renderAdmin();
 renderCart();
}

if(localStorage.getItem("sc_dark")==="true"){
 document.body.classList.add("dark");
}

const savedColor=localStorage.getItem("sc_color");
if(savedColor){
 document.documentElement.style.setProperty("--main",savedColor);
 document.getElementById("colorPicker").value=savedColor;
}

renderAll();
</script>
