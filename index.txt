<script>
const ADMIN_PASSWORD="1234";

const initialProducts=[
 {id:1,name:"هدفون بی‌سیم",category:"دیجیتال",emoji:"🎧"},
 {id:2,name:"ساعت هوشمند",category:"دیجیتال",emoji:"⌚"},
 {id:3,name:"کیبورد گیمینگ",category:"گیمینگ",emoji:"⌨️"},
 {id:4,name:"کفش اسپرت",category:"پوشاک",emoji:"👟"},
 {id:5,name:"چراغ مطالعه",category:"خانه",emoji:"💡"},
 {id:6,name:"دسته بازی",category:"گیمینگ",emoji:"🎮"}
];

function loadProducts(){
 try{
  const saved=localStorage.getItem("shopcity_products");
  return saved===null?initialProducts.map(p=>({...p})):JSON.parse(saved);
 }catch(e){return initialProducts.map(p=>({...p}))}
}
let products=loadProducts();
let activeCategory="همه";

function save(){
 localStorage.setItem("shopcity_products",JSON.stringify(products));
}
function safe(s){
 return String(s).replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
}
function renderCategories(){
 const cats=["همه",...new Set(products.map(p=>p.category))];
 document.getElementById("categories").innerHTML=cats.map(c=>
 `<button class="category ${c===activeCategory?'active':''}" onclick="selectCategory(${JSON.stringify(c)})">${safe(c)}</button>`
 ).join("");
}
function selectCategory(c){
 activeCategory=c;
 renderCategories();
 renderProducts();
}
function renderProducts(){
 const q=document.getElementById("search").value.trim();
 const filtered=products.filter(p=>
 (activeCategory==="همه"||p.category===activeCategory)&&p.name.includes(q)
 );
 document.getElementById("products").innerHTML=filtered.map(p=>`
 <article class="product">
  <div class="product-image">${safe(p.emoji)}</div>
  <small>${safe(p.category)}</small>
  <h3>${safe(p.name)}</h3>
  <div class="unavailable">ناموجود</div>
 </article>`).join("")||'<p>کالایی پیدا نشد.</p>';
}
function renderAdmin(){
 document.getElementById("adminProducts").innerHTML=products.map(p=>`
 <div class="admin-item">
  <span>${safe(p.emoji)} ${safe(p.name)} — ناموجود</span>
  <button class="delete" onclick="deleteProduct(${p.id})">حذف</button>
 </div>`).join("")||"<p>محصولی وجود ندارد.</p>";
}
function addProduct(){
 const name=document.getElementById("newName").value.trim();
 const category=document.getElementById("newCategory").value;
 const emoji=document.getElementById("newEmoji").value.trim()||"📦";
 if(!name){alert("نام محصول را وارد کن.");return}
 products.push({id:Date.now(),name,category,emoji});
 save();
 document.getElementById("newName").value="";
 document.getElementById("newEmoji").value="";
 renderAll();
}
function deleteProduct(id){
 products=products.filter(p=>p.id!==id);
 save();
 renderAll();
}
function toggleAdmin(){
 const panel=document.getElementById("adminPanel");
 panel.classList.toggle("show");
 if(panel.classList.contains("show")){
  if(sessionStorage.getItem("shopcity_admin")==="yes")showAdmin();
  else{
   document.getElementById("login").style.display="block";
   document.getElementById("admin").style.display="none";
  }
 }
}
function loginAdmin(){
 if(document.getElementById("password").value===ADMIN_PASSWORD){
  sessionStorage.setItem("shopcity_admin","yes");
  document.getElementById("loginError").textContent="";
  document.getElementById("password").value="";
  showAdmin();
 }else{
  document.getElementById("loginError").textContent="رمز عبور اشتباه است!";
 }
}
function showAdmin(){
 document.getElementById("login").style.display="none";
 document.getElementById("admin").style.display="block";
 renderAdmin();
}
function logoutAdmin(){
 sessionStorage.removeItem("shopcity_admin");
 document.getElementById("admin").style.display="none";
 document.getElementById("login").style.display="block";
 document.getElementById("password").value="";
}
function toggleTheme(){
 document.body.classList.toggle("dark");
 localStorage.setItem("shopcity_dark",document.body.classList.contains("dark"));
}
function changeColor(color){
 document.documentElement.style.setProperty("--main",color);
 localStorage.setItem("shopcity_color",color);
}
function renderAll(){
 renderCategories();
 renderProducts();
 renderAdmin();
}
if(localStorage.getItem("shopcity_dark")==="true")document.body.classList.add("dark");
const color=localStorage.getItem("shopcity_color");
if(color){
 document.documentElement.style.setProperty("--main",color);
 document.getElementById("colorPicker").value=color;
}
renderAll();
